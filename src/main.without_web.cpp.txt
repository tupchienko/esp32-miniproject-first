#include <Arduino.h>
const int SENSOR_PIN = 6; 
const int RELAY_PIN = 4; 


const int THRESHOLD_DARK = 2400; 
const int THRESHOLD_LIGHT = 2900;

const unsigned long INTERVAL = 200; 
unsigned long previousMillis = 0;  


const float ALPHA = 0.2;            
float filteredValue = 0;          

void setup() {
  Serial.begin(115200);
  
  pinMode(RELAY_PIN, OUTPUT);
  
  digitalWrite(RELAY_PIN, LOW); 
    analogReadResolution(12);

  filteredValue = analogRead(SENSOR_PIN);
}


void loop() {
  unsigned long currentMillis = millis();

  if (currentMillis - previousMillis < INTERVAL) {
    return;
  }
  previousMillis = currentMillis;

  int rawValue = analogRead(SENSOR_PIN);

  filteredValue = ALPHA * rawValue + (1 - ALPHA) * filteredValue;

  if (filteredValue < THRESHOLD_DARK) {
    digitalWrite(RELAY_PIN, HIGH); 
  } else if (filteredValue > THRESHOLD_LIGHT) {
    digitalWrite(RELAY_PIN, LOW); 
  }

  Serial.print(">Raw:");
  Serial.println(rawValue);
  Serial.print(">EMA:");
  Serial.println(filteredValue);
}
